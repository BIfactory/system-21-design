```jsx
function PersonaSwitcher({ role, roles = DEFAULT_PERSONA_ROLES, onChange, hidden }) {
  if (hidden) return null;
  const [open, setOpen] = React.useState(false);
  const current = roles.find(r => r.id === role) || roles[0];
  return (
    <div className="s21-persona-switcher">
      <button type="button" className="s21-persona-switcher__trigger" onClick={() => setOpen(v => !v)} aria-haspopup="menu" aria-expanded={open}>
        {current.icon && <Icon name={current.icon} size={14} />}
        <span>{current.label}</span>
        <Icon name="chevron-down" size={12} />
      </button>
      {open && (
        <div className="s21-persona-switcher__dropdown" role="menu">
          {roles.map(r => (
            <button key={r.id} role="menuitem" type="button" className={"s21-persona-switcher__item " + (r.id === role ? "is-active" : "")} onClick={() => { onChange?.(r.id); setOpen(false); }}>
              {r.icon && <Icon name={r.icon} size={14} />}
              <span>{r.label}</span>
            </button>
          ))}
        </div>
      )}
    </div>
  );
}
```

```js
function getCapacityForWeek(workerId, dateIso) {
  return CAPACITY.filter(c => c.workerId === workerId && c.date === dateIso);
}
function getUtilization(workerId, dateIso) {
  const hours = getCapacityForWeek(workerId, dateIso).reduce((s, c) => s + c.hours, 0);
  return Math.round(hours / 8 * 100);
}
function getUnassignedOperations() {
  // For each active order, each operation: plannedHours - sum(CAPACITY where orderId+operationCode match)
  // Return those with remainingHours > 0
  const active = getActiveOrders();
  const result = [];
  active.forEach(o => {
    (o.operations || []).forEach(op => {
      const assigned = CAPACITY
        .filter(c => c.orderId === o.id && c.operationCode === op.code)
        .reduce((s, c) => s + c.hours, 0);
      const remaining = (op.planned || 0) - assigned;
      if (remaining > 0) result.push({ orderId: o.id, orderCode: o.code, operationCode: op.code, operationName: op.name, remainingHours: remaining });
    });
  });
  return result;
}
```
